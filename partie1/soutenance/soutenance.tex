\section{Modéliser Meowdoku en CNF}

\begin{frame}{Q1 -- Encodage : variables, clauses, correction}
  \begin{columns}[T]
    \column{0.64\textwidth}
    {\small Grille $N\times N$, $N$ régions ($N = M = K$) : un chat par ligne, colonne et région, aucun chat collé (8-voisinage).\par}
    \medskip
    \textbf{Variables.} $x_{i,j}$ vraie ssi chat en $(i,j)$ ; numéro DIMACS $\varphi(i,j) = (i-1)N + j \in \llbracket 1,N^2 \rrbracket$.
    \medskip

    \textbf{Clauses.}
    \begin{itemize}
      \item \textbf{Conflit} $\neg x_a \lor \neg x_b$ : $a,b$ sur la même ligne, colonne, région, ou adjacentes $\Rightarrow$ \emph{au plus un}.
      \item \textbf{Région} $\bigvee_{a \in R_k} x_a$ $\Rightarrow$ \emph{au moins un} par région.
    \end{itemize}
    \medskip
    \textbf{Correction.} Exactement un chat par région, donc $N$ chats ; au plus un par ligne $\Rightarrow$ exactement un (tiroirs), idem colonnes.
    Donc \textbf{modèles de la CNF $=$ solutions du jeu}.
    \medskip

    {\footnotesize Coût : $O(n^4)$ paires testées, $O(n^3)$ clauses en pratique (309 pour la grille du sujet). \texttt{meowdoku.py}\par}
    \column{0.3\textwidth}
    \centering
    \grille[0.4]{\sujet}{} \\ {\footnotesize grille du sujet} \\[4pt]
    \grille[0.4]{\sujet}{3/2,2/5,1/3,5/6,4/4,6/1} \\ {\footnotesize une solution}
  \end{columns}
\end{frame}

\begin{frame}{Q1 -- Générateur et validation}
  \begin{columns}[c]
    \column{0.64\textwidth}
    \textbf{Solution plantée} (\texttt{generator.py}) :
    \begin{enumerate}
      \item placer $N$ chats valides (backtracking aléatoire ; impossible pour $N = 2,3$) ;
      \item chaque chat est la graine d'une région ; on rattache au hasard une case libre voisine (4-voisinage) jusqu'à remplir la grille.
    \end{enumerate}
    Chaque région contient exactement un chat : la grille a au moins une solution (pas forcément unique).
    \medskip

    \textbf{Validation.} \texttt{is\_valid} vérifie chaque solution directement avec les règles du jeu, sans passer par la CNF.
    \column{0.3\textwidth}
    \centering \grille{\genere}{1/3,2/1,3/6,4/4,5/2,6/5} \\
    \footnotesize $N=6$, graine 1
  \end{columns}
\end{frame}

\section{DIMACS et DPLL}

\begin{frame}[fragile]{Q2 -- Lecture DIMACS et DPLL : objets et opérations}
  \begin{columns}[T]
    \column{0.44\textwidth}
    \begin{alertblock}{Objets}
      \small
      \begin{tabular}{@{}ll@{}}
        littéral & entier : $k = x_k$, $-k = \neg x_k$ \\
        clause & \texttt{frozenset} de littéraux \\
        formule & \texttt{set} de clauses \\
        valuation & \texttt{set} des littéraux vrais \\
      \end{tabular}
    \end{alertblock}
    \textbf{Lecture} (\texttt{dimacs.py}) :\par\smallskip
    \begin{minipage}[c]{0.27\linewidth}
\begin{lstlisting}[language={},aboveskip=0pt,belowskip=0pt]
p cnf 3 3
-1 2 0
-1 -2 3 0
1 0
\end{lstlisting}
    \end{minipage}%
    $\rightarrow$\hspace{2pt}%
    \begin{minipage}[c]{0.62\linewidth}
\begin{lstlisting}[aboveskip=0pt,belowskip=0pt,basicstyle=\ttfamily\scriptsize]
{frozenset({-1, 2}),
 frozenset({-1, -2, 3}),
 frozenset({1})}
\end{lstlisting}
    \end{minipage}
    \medskip

    {\footnotesize $\Rightarrow$ \textbf{Conséquence} : les doublons sont éliminés.\par
    Une clause se termine au \texttt{0}, pas en fin de ligne.\par}

    \column{0.53\textwidth}
    \begin{block}{Opérations (\texttt{dpll.py})}
      \small\renewcommand{\arraystretch}{1.15}
      \begin{tabular}{@{}>{\bfseries}l>{\raggedright\arraybackslash}p{0.66\linewidth}@{}}
        \multicolumn{2}{@{}l}{\cellcolor{blue!12}\textit{Tester}} \\
        arrêt & $\square \in F \Rightarrow$ UNSAT ;\; $F = \emptyset \Rightarrow$ SAT \\
        \rowcolor{black!5} unitaire & clause de taille 1 \\
        pur & $l$ présent, $-l$ absent \\
        \multicolumn{2}{@{}l}{\cellcolor{blue!12}\textit{Transformer}} \\
        $F[l \leftarrow 1]$ & retirer les clauses $\ni l$, retirer $-l$ des autres $\to$ \emph{nouvelle} formule \\
        \rowcolor{black!5} décision & $l = $ \texttt{choix(F)} (heuristique $\to$ Q4) \\
        retour & reprendre l'ancienne formule \\
      \end{tabular}
    \end{block}
    {\footnotesize \texttt{dpll\_sat} : statut ; \texttt{dpll} : statut $+$ valuation.\par}
  \end{columns}
\end{frame}

\begin{frame}{Q4 -- Heuristiques implémentées (\texttt{heuristics.py})}
  \begin{center}\small
  \begin{tabular}{@{}l>{\raggedright\arraybackslash}p{0.56\textwidth}>{\raggedright\arraybackslash}p{0.27\textwidth}@{}}
    \toprule
    & Critère de choix & Propriétés \\
    \midrule
    H0 & variable et valeur tirées au hasard & \\
    H1 & variable libre de plus petit numéro, mise à vrai & \\
    H2s, H2d & variable la plus fréquente & H2d dynamique \\
    H3d & littéral le plus fréquent & littéral, dynamique \\
    H4d & variable la plus fréquente dans les clauses minimales & taille des clauses, dynamique \\
    H7s, H7d & littéral de plus grand poids $\sum 2^{-|C|}$ & littéral, taille, H7d dynamique \\
    \textbf{H8} & \textbf{plus courte clause positive}, une variable mise à vrai & taille, dynamique (nous) \\
    \bottomrule
  \end{tabular}
  \end{center}
  {\footnotesize Contraintes du sujet respectées : une heuristique sur un littéral, une sur la taille des clauses, une dynamique.\par}
  \smallskip

  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Statique} : score calculé une fois sur la formule initiale $F_0$.\par
    \textbf{Dynamique} : score recalculé sur $F$ à chaque décision.
    \column{0.46\textwidth}
    \textbf{Vérification} : même statut que l'algorithme 2 sur les instances de la Q2, modèles valides.
  \end{columns}
\end{frame}

\begin{frame}{Q4 -- H8 : une heuristique pour Meowdoku}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \begin{alertblock}{Observation}
      \small Dans notre encodage, les seules clauses \textbf{entièrement positives} sont les clauses de région « au moins un chat ».
    \end{alertblock}
    \textbf{H8} :
    \begin{enumerate}\small
      \item prendre la plus courte clause positive : la région qui a \textbf{le moins de cases possibles} ;
      \item y choisir la variable la plus fréquente ;
      \item la mettre à \textbf{1} : placer un chat ;
      \item plus de clause positive $\Rightarrow$ H7d.
    \end{enumerate}
    \column{0.41\textwidth}
    \textbf{Pourquoi ça marche}
    \begin{itemize}\small
      \item Un chat posé rend unitaires tous ses conflits $\neg x_a \lor \neg x_b$ : sa ligne, sa colonne, sa région et ses voisines tombent à 0 par propagation.
      \item Les autres heuristiques mettent les cases à \textbf{0} (plus fréquentes en négatif) : une seule case éliminée.
      \item Région la plus contrainte d'abord : on se trompe moins, et on le voit plus vite.
    \end{itemize}
    \smallskip
    {\footnotesize \textbf{Limite} : dépend de l'encodage, pas générique. Résultats $\to$ Q5.\par}
  \end{columns}
\end{frame}

\begin{frame}{Q5 -- Comparaison des heuristiques : analyse}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \begin{block}{Une heuristique améliore-t-elle DPLL ?}
      \small Oui, mais \textbf{ça dépend du problème}.
      \begin{itemize}
        \item 3-SAT : les dynamiques font 5 à 10 fois moins de décisions que le hasard ; H4d est la meilleure.
        \item Meowdoku : H2d, H3d, H4d, H7d mettent les cases à \textbf{0} dans 100~\% des décisions (variables bien plus souvent négatives) : presque aucune propagation.
        \item H8 met une case à \textbf{1} : 9 décisions au lieu de 169, seule à résoudre les 33 UNSAT.
      \end{itemize}
    \end{block}
    \column{0.46\textwidth}
    \begin{block}{Le dynamique vaut-il son coût ?}
      \small Oui sur Meowdoku : l'heuristique prend 45 à 54~\% du CPU (25~\% en statique), mais sur les grilles plantées H2s fait 739 décisions et H2d 204.
    \end{block}
    \begin{block}{Conformes à nos attentes ?}
      \small Sur 3-SAT oui. Sur Meowdoku non : H2d, H3d, H7d font même pire que le hasard sur les petites grilles ($n=10$ : 88 décisions contre 17).
    \end{block}
    {\footnotesize \textbf{Limites} : une exécution par cas, peu d'instances par taille : des tendances.\par}
  \end{columns}
\end{frame}
